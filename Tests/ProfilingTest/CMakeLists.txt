cmake_minimum_required(VERSION 3.16)
if(NOT TARGET FeiLib)
    project(FeiLibProfilingValidation LANGUAGES CXX)
    enable_testing()
endif()
find_package(Threads REQUIRED)
get_filename_component(PROFILE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../FeiLib" ABSOLUTE)
if(TARGET FeiLib)
    set(PROFILE_LIBRARY FeiLib)
else()
    add_library(ProfilerUnderTest STATIC "${PROFILE_ROOT}/src/Profiling/FProfiler.cpp")
    target_compile_features(ProfilerUnderTest PUBLIC cxx_std_20)
    target_compile_definitions(ProfilerUnderTest PUBLIC _F_EXPORT=1)
    target_include_directories(ProfilerUnderTest PUBLIC "${PROFILE_ROOT}/include")
    target_link_libraries(ProfilerUnderTest PUBLIC Threads::Threads)
    set(PROFILE_LIBRARY ProfilerUnderTest)
endif()
add_executable(FeiLibProfilingTests test.cpp)
target_link_libraries(FeiLibProfilingTests PRIVATE ${PROFILE_LIBRARY} Threads::Threads)
add_test(NAME FeiLib.Profiling COMMAND FeiLibProfilingTests)
set_tests_properties(FeiLib.Profiling PROPERTIES TIMEOUT 30)

add_executable(FeiLibProfilingDisabledTests test.cpp "${PROFILE_ROOT}/src/Profiling/FProfiler.cpp")
target_compile_features(FeiLibProfilingDisabledTests PRIVATE cxx_std_20)
target_compile_definitions(FeiLibProfilingDisabledTests PRIVATE _F_EXPORT=1 FEILIB_ENABLE_PROFILING=0)
target_include_directories(FeiLibProfilingDisabledTests PRIVATE "${PROFILE_ROOT}/include")
target_link_libraries(FeiLibProfilingDisabledTests PRIVATE Threads::Threads)
add_test(NAME FeiLib.ProfilingCompiledOut COMMAND FeiLibProfilingDisabledTests)
set_tests_properties(FeiLib.ProfilingCompiledOut PROPERTIES TIMEOUT 30)
