cmake_minimum_required(VERSION 3.16)
project(ScriptBindingTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
find_package(OpenSSL REQUIRED)
add_subdirectory(../AsyncTest async)
get_filename_component(FEILIB_ROOT "../../FeiLib" ABSOLUTE)
add_subdirectory(${FEILIB_ROOT}/3rd/quickjs quickjs)
file(GLOB SCRIPT_IMPLEMENTATION CONFIGURE_DEPENDS "${FEILIB_ROOT}/src/Script/*.cpp")
target_sources(FeiLibAsyncUnderTest PRIVATE ${SCRIPT_IMPLEMENTATION})
target_include_directories(FeiLibAsyncUnderTest PUBLIC "${FEILIB_ROOT}/3rd/json/include")
target_include_directories(FeiLibAsyncUnderTest PRIVATE "${FEILIB_ROOT}/src/Script")
target_compile_definitions(FeiLibAsyncUnderTest PUBLIC FEILIB_ENABLE_SCRIPT=1)
target_link_libraries(FeiLibAsyncUnderTest PRIVATE FeiQuickJS)
add_executable(script_binding_test test.cpp)
target_include_directories(script_binding_test PRIVATE "${FEILIB_ROOT}/src/Script" "${FEILIB_ROOT}/3rd/quickjs")
target_link_libraries(script_binding_test PRIVATE FeiLibAsyncUnderTest)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME Script.FeilibHttpBinding COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/run.py $<TARGET_FILE:script_binding_test>)
set_tests_properties(Script.FeilibHttpBinding PROPERTIES TIMEOUT 20)
add_library(ScriptServiceUnderTest STATIC
    ../../Server/ScriptRuntime/ScriptService.cpp ../../Server/ScriptRuntime/ModuleBundle.cpp
    ../../Server/ScriptRuntime/VersionStore.cpp ../../Server/ScriptRuntime/ScriptVersions.cpp)
target_include_directories(ScriptServiceUnderTest PUBLIC ../../Server/ScriptRuntime)
target_link_libraries(ScriptServiceUnderTest PUBLIC FeiLibAsyncUnderTest OpenSSL::Crypto)
add_executable(script_service_test service_test.cpp)
target_link_libraries(script_service_test PRIVATE ScriptServiceUnderTest)
add_test(NAME Script.BackgroundLifecycle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/service_run.py $<TARGET_FILE:script_service_test>)
set_tests_properties(Script.BackgroundLifecycle PROPERTIES TIMEOUT 30)
add_executable(script_versions_test versions_test.cpp)
target_link_libraries(script_versions_test PRIVATE ScriptServiceUnderTest)
add_test(NAME Script.VersionActivation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/versions_run.py $<TARGET_FILE:script_versions_test>)
set_tests_properties(Script.VersionActivation PROPERTIES TIMEOUT 45)
add_executable(script_engine_test engine_test.cpp)
target_link_libraries(script_engine_test PRIVATE FeiLibAsyncUnderTest)
add_test(NAME Script.EngineConcurrency COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/run.py $<TARGET_FILE:script_engine_test>)
set_tests_properties(Script.EngineConcurrency PROPERTIES TIMEOUT 25)
add_executable(script_binary_test binary_test.cpp)
target_link_libraries(script_binary_test PRIVATE FeiLibAsyncUnderTest)
add_test(NAME Script.BinaryHttp COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/binary_run.py $<TARGET_FILE:script_binary_test>)
set_tests_properties(Script.BinaryHttp PROPERTIES TIMEOUT 35)
add_executable(script_memory_test memory_test.cpp)
target_link_libraries(script_memory_test PRIVATE FeiLibAsyncUnderTest)
add_test(NAME Script.MemoryLimit COMMAND script_memory_test)
set_tests_properties(Script.MemoryLimit PROPERTIES TIMEOUT 20)
