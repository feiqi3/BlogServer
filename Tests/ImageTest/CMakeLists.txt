cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ServerImageValidation LANGUAGES C CXX)
    enable_testing()
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    add_subdirectory(../AsyncTest async)
    set(IMAGE_ASYNC_LIBRARY FeiLibAsyncUnderTest)
else()
    set(IMAGE_ASYNC_LIBRARY FeiLib)
endif()
get_filename_component(BLOG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET sqlite3)
    add_subdirectory("${BLOG_ROOT}/Server/3rd/sqlite3" sqlite)
endif()
add_executable(ServerImageTests image_test.cpp
    "${BLOG_ROOT}/Server/src/Utils/ImageInfo.cpp"
    "${BLOG_ROOT}/Server/src/Utils/ImageRequester.cpp"
    "${BLOG_ROOT}/Server/src/Service/ImageService.cpp"
    "${BLOG_ROOT}/Server/src/DAO/DataBaseOperation.cpp"
    "${BLOG_ROOT}/Server/src/DAO/PhotoQuery.cpp")
target_include_directories(ServerImageTests PRIVATE "${BLOG_ROOT}/Server/src"
    "${BLOG_ROOT}/Server/3rd/stb" "${BLOG_ROOT}/FeiLib/3rd/reflect" "${BLOG_ROOT}/Tests/AsyncTest")
target_link_libraries(ServerImageTests PRIVATE ${IMAGE_ASYNC_LIBRARY} sqlite3)
if(WIN32)
    find_package(OpenSSL REQUIRED)
    file(GLOB IMAGE_OPENSSL_DLLS "${OPENSSL_INCLUDE_DIR}/../bin/*ssl*.dll" "${OPENSSL_INCLUDE_DIR}/../bin/*crypto*.dll")
    foreach(IMAGE_DLL IN LISTS IMAGE_OPENSSL_DLLS)
        add_custom_command(TARGET ServerImageTests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${IMAGE_DLL}" "$<TARGET_FILE_DIR:ServerImageTests>")
    endforeach()
    add_custom_command(TARGET ServerImageTests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:sqlite3>" "$<TARGET_FILE_DIR:ServerImageTests>")
endif()
if(IMAGE_ASYNC_LIBRARY STREQUAL "FeiLib")
    target_compile_definitions(ServerImageTests PRIVATE IMAGE_TEST_FULL_SERVER=1)
    add_custom_command(TARGET ServerImageTests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_BINARY_DIR}/Bin/$<CONFIG>" "$<TARGET_FILE_DIR:ServerImageTests>")
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME Server.ImageMetadata COMMAND ServerImageTests metadata)
add_test(NAME Server.ImageWorkflow COMMAND ${Python3_EXECUTABLE} "${BLOG_ROOT}/Tests/AsyncTest/run_tests.py" "$<TARGET_FILE:ServerImageTests>" --groups image)
set_tests_properties(Server.ImageMetadata PROPERTIES TIMEOUT 30)
set_tests_properties(Server.ImageWorkflow PROPERTIES TIMEOUT 90)
